import pytest
from subject import merge_intervals


def test_empty_intervals():
    assert merge_intervals([]) == []


def test_single_interval():
    assert merge_intervals([[1, 5]]) == [[1, 5]]


def test_non_overlapping_intervals():
    assert merge_intervals([[1, 2], [4, 5], [7, 10]]) == [[1, 2], [4, 5], [7, 10]]


def test_unsorted_non_overlapping_intervals():
    assert merge_intervals([[7, 10], [1, 2], [4, 5]]) == [[1, 2], [4, 5], [7, 10]]


def test_overlapping_intervals():
    assert merge_intervals([[1, 3], [2, 6], [8, 10], [15, 18]]) == [[1, 6], [8, 10], [15, 18]]


def test_touching_intervals():
    assert merge_intervals([[1, 4], [4, 5]]) == [[1, 5]]


def test_containing_intervals():
    assert merge_intervals([[1, 10], [2, 5], [3, 8]]) == [[1, 10]]


def test_exact_duplicates():
    assert merge_intervals([[1, 5], [1, 5], [1, 5]]) == [[1, 5]]


def test_point_intervals():
    assert merge_intervals([[1, 1], [2, 2], [3, 3]]) == [[1, 1], [2, 2], [3, 3]]
    assert merge_intervals([[1, 1], [1, 1]]) == [[1, 1]]
    assert merge_intervals([[1, 2], [2, 2], [2, 3]]) == [[1, 3]]


def test_negative_intervals():
    assert merge_intervals([[-10, -5], [-7, -2], [0, 2], [-1, 1]]) == [[-10, -2], [-1, 2]]


def test_all_merge_into_single():
    assert merge_intervals([[1, 2], [2, 3], [3, 4], [4, 5]]) == [[1, 5]]


def test_reversed_interval_raises_value_error():
    with pytest.raises(ValueError, match="reversed interval"):
        merge_intervals([[5, 1]])


def test_mixed_reversed_interval_raises_value_error():
    with pytest.raises(ValueError, match="reversed interval"):
        merge_intervals([[1, 3], [2, 6], [8, 7], [15, 18]])
