import pytest
import math
from subject import merge_intervals


def test_merge_normal_overlapping():
    assert merge_intervals([[1, 3], [2, 6], [8, 10], [15, 18]]) == [[1, 6], [8, 10], [15, 18]]


def test_merge_touching_intervals():
    assert merge_intervals([[1, 4], [4, 5]]) == [[1, 5]]


def test_merge_containing_intervals():
    assert merge_intervals([[1, 5], [2, 4]]) == [[1, 5]]


def test_merge_unsorted_intervals():
    assert merge_intervals([[8, 10], [1, 3], [2, 6], [15, 18]]) == [[1, 6], [8, 10], [15, 18]]


def test_merge_identical_intervals():
    assert merge_intervals([[1, 3], [1, 3], [2, 2]]) == [[1, 3]]


def test_boundary_single_interval():
    assert merge_intervals([[1, 5]]) == [[1, 5]]


def test_boundary_empty_list():
    assert merge_intervals([]) == []


def test_boundary_point_intervals():
    assert merge_intervals([[1, 1], [2, 2], [1, 2]]) == [[1, 2]]


def test_invalid_input_reversed_interval():
    with pytest.raises(ValueError):
        merge_intervals([[5, 1]])


def test_invalid_input_one_reversed_in_many():
    with pytest.raises(ValueError):
        merge_intervals([[1, 3], [5, 4], [8, 10]])
